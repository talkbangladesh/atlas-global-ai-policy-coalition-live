function institutionalDocumentPage(kind) {
  const reader = kind === 'reader';
  const title = reader ? 'Reader Charter' : 'Committee Rulebook';
  const eyebrow = reader ? 'READER CHARTER' : 'COMMITTEE RULEBOOK';
  const intro = reader ? 'A clear guide to engaging with the Coalition’s research, policy perspectives and institutional publications.' : 'A framework for understanding the purpose, responsibilities and working principles of the Coalition’s committees.';
  const sections = reader ? [['Purpose','The Reader Charter sets out how Coalition publications and contributions are intended to be read: as evidence-informed work that welcomes careful interpretation, constructive scrutiny and diverse perspectives.'],['Evidence and context','Readers are encouraged to consider the source, scope, methods and context of each publication. Institutional contributions should be understood alongside the author’s stated position and the material’s publication date.'],['Respectful dialogue','The Coalition values rigorous disagreement conducted with respect for human dignity, cultural difference and the rights of participating communities.'],['Responsible use','Readers should not present a Coalition publication as legal, regulatory or institutional advice unless the publication expressly says so.']]:[['Purpose and scope','The Committee Rulebook describes the principles that guide committee participation, collaboration and responsible institutional conduct within the Coalition.'],['Integrity and independence','Committee participants are expected to act with professional integrity, disclose relevant conflicts and respect the independence of participating experts and institutions.'],['Responsibilities','Committees support research collaboration, policy dialogue, knowledge exchange and institutional capacity-building within their authorized scope.'],['Authorization and review','Leadership appointments, committee mandates and formal decisions remain subject to authorized institutional review.']];
  return `<main><section class="page-hero"><div class="wrap"><span class="kicker">${eyebrow}</span><h1>${title}</h1><p>${intro}</p></div></section><section class="wrap document-page"><div class="document-lead"><span class="status-pill">ATLAS AI INSTITUTE · COALITION FRAMEWORK</span><h2>${reader ? 'Read with context. Engage with care.' : 'Shared principles for responsible committee work.'}</h2><p>${reader ? 'The Coalition brings together research, policy and professional perspectives from diverse contexts. This charter helps readers approach that work with clarity and care.' : 'The Coalition’s committees are collaborative structures designed to support the mission of inclusive, responsible and globally informed AI governance.'}</p></div><div class="document-sections">${sections.map(([heading,body],i)=>`<section class="document-section"><span>0${i+1}</span><h3>${heading}</h3><p>${body}</p></section>`).join('')}</div><div class="document-note"><b>Institutional note</b><p>This page provides the current public framework. Any formal amendment, appointment or binding institutional decision remains subject to authorized review.</p></div><div class="document-actions">${button('Return to About','/about/')} ${button(reader ? 'View Committee Rulebook' : 'View Reader Charter', reader ? '/committee-rulebook/' : '/reader-charter/', true)}</div></section></main>`;
}
