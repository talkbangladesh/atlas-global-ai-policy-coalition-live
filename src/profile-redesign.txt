function profile(slug) { const m = state.members.find(x => x.slug === slug); if (!m) return missing(); const links = Object.entries(m.social_links || {}).filter(([, v]) => v); const expertise = Array.isArray(m.professional_expertise) ? m.professional_expertise.filter(Boolean) : []; const bioParagraphs = String(m.biography || '').split(/\n+/).map(x => x.trim()).filter(Boolean); const profileFacts = []; if (expertise.length) profileFacts.push(['✦','Expertise',expertise.join(', ')]); if (m.country) profileFacts.push(['⌖','Country',m.country]); if (m.created_at) profileFacts.push(['◷','Member since',String(m.created_at).slice(0,4)]); const details = []; if (m.institution) details.push(['Organization',m.institution]); if (m.designation) details.push(['Role',m.designation]); if (m.qualification) details.push(['Qualification',m.qualification]); const others = state.members.filter(x => x.slug !== m.slug).sort((a,b) => (a.display_order||0)-(b.display_order||0)).slice(0,6); const bioMarkup = bioParagraphs.length ? bioParagraphs.map((p,i)=>`<p class="${i===0?'bio-lead':''}">${esc(p)}</p>`).join('') : '<div class="profile-empty">Biography will be added here when available.</div>'; const detailMarkup = details.length ? details.map(([label,value])=>`<div><dt>${esc(label)}</dt><dd>${esc(value)}</dd></div>`).join('') : '<div class="profile-empty">Professional information will be added here when available.</div>'; const linksMarkup = links.map(([label,url])=>`<a class="profile-social-link" href="${esc(url)}" target="_blank" rel="noopener">${esc(label)} ↗</a>`).join(''); return `<main class="member-profile-page"><div class="wrap profile-breadcrumbs"><a href="/">Home</a><span>/</span><a href="/members/">Members</a><span>/</span><span>${esc(m.full_name)}</span><a class="profile-back" href="/members/">← Back to members</a></div><section class="wrap profile-layout"><aside class="profile-side"><div class="profile-card"><div class="profile-banner" aria-hidden="true"></div><div class="profile-avatar">${m.photo ? `<img src="${esc(m.photo)}" alt="${esc(m.photo_alt || m.full_name)}">` : `<span>${esc(initials(m.full_name))}</span>`}</div><div class="profile-card-body"><h1>${esc(m.full_name)}</h1><span class="role-badge">${esc(m.membership_type || 'Member')}</span>${m.designation || m.institution ? `<p class="profile-role-line">${[m.designation,m.institution].filter(Boolean).map(esc).join(' · ')}</p>` : ''}${m.country ? `<p class="profile-country-line">⌖ ${esc(m.country)}</p>` : ''}<div class="profile-actions">${(m.social_links||{}).linkedin ? `<a class="button profile-linkedin" href="${esc(m.social_links.linkedin)}" target="_blank" rel="noopener">Connect on LinkedIn ↗</a>` : ''}<button class="icon-button" id="profile-copy" type="button" aria-label="Copy profile link" title="Copy profile link">⧉</button><button class="icon-button" id="profile-print" type="button" aria-label="Print profile" title="Print profile">⎙</button><button class="icon-button" id="profile-share" type="button" aria-label="Share profile" title="Share profile">↗</button></div>${linksMarkup ? `<div class="profile-socials">${linksMarkup}</div>` : ''}</div></div><div class="profile-facts">${profileFacts.map(([icon,label,value])=>`<div class="fact-chip"><span aria-hidden="true">${icon}</span><div><small>${esc(label)}</small><b>${esc(value)}</b></div></div>`).join('')}</div></aside><div class="profile-main"><section class="profile-section biography-card"><div class="section-label">PROFILE</div><h2>Biography</h2><div class="profile-bio-text ${bioParagraphs.length>1?'is-collapsed':''}">${bioMarkup}</div>${bioParagraphs.length>1 ? '<button class="read-more-button" id="profile-bio-toggle" type="button">Read more</button>' : ''}</section><section class="profile-section"><div class="section-label">BACKGROUND</div><h2>Professional information</h2><dl class="profile-definition-list">${detailMarkup}${expertise.length ? `<div><dt>Expertise</dt><dd><span class="tag-list">${expertise.map(x=>`<span class="profile-tag">${esc(x)}</span>`).join('')}</span></dd></div>` : ''}</dl></section>${expertise.length ? `<section class="profile-section"><div class="section-label">FOCUS</div><h2>Areas of focus</h2><div class="tag-list">${expertise.map(x=>`<span class="profile-tag">${esc(x)}</span>`).join('')}</div></section>` : ''}<section class="profile-section other-members"><div class="section-label">THE COALITION</div><h2>Other Coalition members</h2><div class="other-members-track">${others.map(x=>`<a class="other-member-card" href="/members/${esc(x.slug)}/"><span class="other-member-avatar">${x.photo ? `<img src="${esc(x.photo)}" alt="${esc(x.photo_alt || x.full_name)}">` : esc(initials(x.full_name))}</span><strong>${esc(x.full_name)}</strong><small>${esc(x.designation || x.membership_type || 'Member')}</small></a>`).join('')}</div></section></div></section></main>`; }
